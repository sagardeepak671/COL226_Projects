floatmatrix 3,3 a := (3,3[[1.,33.,3.],[4.,22.,6.],[7.,8.,9.]]) ;
float trace:= 0.;


for (i := 0,i<3,1) {
    trace:= trace + a[i,i];
};

Print(trace);

float det := determinant(a);
float D := (trace * trace) - (4. * det) ; 
 
Print(det);
if( D >= 0.) then {
    float sqrt_D := 0.0;
    float guess := det / 2.;
    for (i := 1, i < 10, 1) {
        guess := (guess + det / guess) / 2.;
    };
    sqrt_D := guess;
    float e1 := (trace + sqrt_D) / 2.;
    float e2 := (trace - sqrt_D) / 2.;
    Print(e1);
    Print(e2);
}else {
    bool result := false;
    Print(result);
};
